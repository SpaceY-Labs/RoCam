### `shared/types/mask.ts`

**Purpose:** Type definitions for mask/annotation entities.

**Implementation Notes:**
- Stored as sub-subcollection: `projects/{projectId}/images/{imageId}/masks/{maskId}`
- `polygon` format: array of rings, first is outer boundary, rest are holes
- Each ring is array of `{x, y}` points in image coordinates
- `boundingBox` precomputed for spatial queries and rendering optimization
- `area` in pixels, used for sorting and filtering small masks
- Used by: `mask.service.ts`, `mask.model.ts`, `useMasks.ts`, `LabelingCanvas.tsx`, exporters

```typescript
import { Timestamp } from './common';

  createdBy: string;            // userId
  createdAt: Timestamp;
  updatedAt: Timestamp;
}

  type: 'polygon';              // Only polygon supported for now
```

### `shared/types/assignment.ts`

**Purpose:** Type definitions for work assignment operations.

**Implementation Notes:**
- Single unified endpoint handles all assignment strategies
- `count` strategy: take N random/oldest/newest from pool
- `images` strategy: assign specific image IDs
- `unassign` strategy: release images from a user
- `rebalance` strategy: redistribute among users
- Lock refresh triggered after any assignment change
- Used by: `assignment.service.ts`, `assignment.routes.ts`, `useAssignImages.ts`, `AssignmentPanel.tsx`

```typescript
type AssignmentStrategy = 'count' | 'images' | 'unassign' | 'rebalance';

interface AssignRequest {
  assignTo?: string | null;     // Target userId (null = back to pool)
  strategy: AssignmentStrategy;
  
  // For strategy: "count"
  count?: number;               // How many to assign
  
  // For strategy: "images"  
  imageIds?: string[];          // Specific images to assign
  
  // For strategy: "unassign"
  unassignFrom?: string;        // User to unassign from
  unassignFilter?: 'all' | 'unlabeled' | 'assigned';
  
  // For strategy: "rebalance"
  distributeAmong?: string[];   // Users to distribute to
  distributeMode?: 'equal' | 'weighted';
  weights?: Record<string, number>;
  
  // Common filters
  filters?: {
    status?: ImageStatus[];
    currentAssignee?: string | null;
  };
  
  priority?: 'random' | 'oldest' | 'newest';
}

interface AssignResponse {
  success: boolean;
  assigned: Record<string, number>;  // userId → count assigned
  summary: {
    totalMoved: number;
    poolRemaining: number;
  };
}

// Lock configuration (from constants)
interface LockConfig {
  threshold: number;            // 10 - images below this unlocked
  durationMs: number;           // 24 hours
}
```

```typescript
export const PROJECT_ROLES = {
  OWNER: 'owner',
  ADMIN: 'admin',
  LABELER: 'labeler',
} as const;

export type ProjectRole = typeof PROJECT_ROLES[keyof typeof PROJECT_ROLES];

// Higher index = more permissions
export const PROJECT_ROLE_HIERARCHY: ProjectRole[] = [
  'labeler',
  'admin', 
  'owner'
];

/**
 * Check if userRole has at least requiredRole permissions
 * @example hasPermission('admin', 'labeler') // true
 * @example hasPermission('labeler', 'admin') // false
 */
export function hasPermission(
  userRole: ProjectRole,
  requiredRole: ProjectRole
): boolean {
  const userLevel = PROJECT_ROLE_HIERARCHY.indexOf(userRole);
  const requiredLevel = PROJECT_ROLE_HIERARCHY.indexOf(requiredRole);
  return userLevel >= requiredLevel;
}
```

### `shared/validation/mask.schema.ts`

**Purpose:** Zod validation schemas for mask operations.

**Implementation Notes:**
- Point coordinates can be float (subpixel precision from SAM)
- Polygon must have at least 3 points to be valid
- Area computed client-side before save
- Used by: `labeling.routes.ts`, `useSaveMasks.ts`

```typescript
import { z } from 'zod';

const pointSchema = z.object({
  x: z.number(),
  y: z.number(),
});

const bboxSchema = z.object({
  x: z.number(),
  y: z.number(),
  w: z.number().positive(),
  h: z.number().positive(),
});

const polygonRingSchema = z.array(pointSchema).min(3);

const maskDataSchema = z.object({
  type: z.literal('polygon'),
  polygon: z.array(polygonRingSchema).min(1),  // At least outer ring
});

const maskSourceSchema = z.enum([
  'sam3_auto',
  'sam3_click', 
  'sam3_semantic',
  'manual',
]);

export const createMaskSchema = z.object({
  classId: z.string().min(1),
  data: maskDataSchema,
  boundingBox: bboxSchema,
  area: z.number().nonnegative(),
  source: maskSourceSchema,
});

export const saveMasksSchema = z.object({
  masks: z.array(createMaskSchema),  // Can be empty (clear all masks)
});

export type CreateMaskInput = z.infer<typeof createMaskSchema>;
export type SaveMasksInput = z.infer<typeof saveMasksSchema>;
```

### `shared/validation/assignment.schema.ts`

**Purpose:** Zod validation schemas for assignment operations.

**Implementation Notes:**
- Complex schema with conditional requirements based on strategy
- Uses `.refine()` for cross-field validation
- Each strategy has required fields checked at runtime
- Used by: `assignment.routes.ts`, `useAssignImages.ts`

```typescript
import { z } from 'zod';

const strategySchema = z.enum(['count', 'images', 'unassign', 'rebalance']);

export const assignRequestSchema = z.object({
  assignTo: z.string().nullable().optional(),
  strategy: strategySchema,
  
  // Strategy: count
  count: z.number().int().positive().optional(),
  
  // Strategy: images
  imageIds: z.array(z.string()).optional(),
  
  // Strategy: unassign
  unassignFrom: z.string().optional(),
  unassignFilter: z.enum(['all', 'unlabeled', 'assigned']).optional(),
  
  // Strategy: rebalance
  distributeAmong: z.array(z.string()).optional(),
  distributeMode: z.enum(['equal', 'weighted']).optional(),
  weights: z.record(z.string(), z.number().positive()).optional(),
  
  // Filters
  filters: z.object({
    status: z.array(z.enum(['unlabeled', 'assigned', 'in_progress', 'labeled'])).optional(),
    currentAssignee: z.string().nullable().optional(),
  }).optional(),
  
  priority: z.enum(['random', 'oldest', 'newest']).optional(),
})
.refine((data) => {
  // Validate strategy-specific required fields
  switch (data.strategy) {
    case 'count':
      return data.count !== undefined && data.count > 0;
    case 'images':
      return data.imageIds !== undefined && data.imageIds.length > 0;
    case 'unassign':
      return data.unassignFrom !== undefined;
    case 'rebalance':
      return data.distributeAmong !== undefined && data.distributeAmong.length > 0;
    default:
      return false;
  }
}, {
  message: 'Invalid strategy configuration - check required fields',
});

export type AssignRequestInput = z.infer<typeof assignRequestSchema>;
```

```typescript
/**

export function generatePath(
  route: string,
  params: Record<string, string>
```

### `frontend/src/api/client.ts`

**Purpose:** Base HTTP client with auth and error handling.

**Implementation Notes:**
- Imports: `env` for API URL, Firebase auth for token
- Token refreshed automatically via Firebase SDK
- All other `*.api.ts` files import and use this client
- Handles 401 by clearing auth state
- Used by: all API modules

```typescript
import { env } from '@/config/env';
import { auth } from '@/config/firebase';
import { ApiError } from '@shared/types';

class ApiClient {
  private baseUrl: string;
  
  constructor() {
    this.baseUrl = env.apiUrl;
  }
  
  /**
   * Get current auth token from Firebase
   * Returns null if not authenticated
   */
  private async getToken(): Promise<string | null> {
    const user = auth.currentUser;
    if (!user) return null;
    return user.getIdToken();
  }
  
  /**
   * Make HTTP request with auth
   * Automatically includes Authorization header
   * Throws ApiError on non-2xx response
   */
  async request<T>(endpoint: string, options: RequestInit = {}): Promise<T> {
    const token = await this.getToken();
    
    const response = await fetch(`${this.baseUrl}${endpoint}`, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        ...(token && { Authorization: `Bearer ${token}` }),
        ...options.headers,
      },
    });
    
    if (!response.ok) {
      const error = await response.json();
      throw new ApiError(error.code, error.message, response.status);
    }
    
    return response.json();
  }
  
  // Convenience methods
  get<T>(endpoint: string): Promise<T> {
    return this.request<T>(endpoint, { method: 'GET' });
  }
  
  post<T>(endpoint: string, body?: unknown): Promise<T> {
    return this.request<T>(endpoint, {
      method: 'POST',
      body: body ? JSON.stringify(body) : undefined,
    });
  }
  
  patch<T>(endpoint: string, body?: unknown): Promise<T> {
    return this.request<T>(endpoint, {
      method: 'PATCH',
      body: body ? JSON.stringify(body) : undefined,
    });
  }
  
  delete<T>(endpoint: string): Promise<T> {
    return this.request<T>(endpoint, { method: 'DELETE' });
  }
}

export const apiClient = new ApiClient();
```

### `frontend/src/hooks/mutations/useUploadImages.ts`

**Purpose:** Image upload mutation with progress tracking.

**Implementation Notes:**
- Imports: `imagesApi` for API calls, `useUIStore` for toasts
- Three-phase upload: get URLs → parallel upload → confirm
- Progress tracked per-file for UI feedback
- AbortController for cancel support
- Used by: `ImageUploader.tsx`

```typescript
import { useState, useRef, useCallback } from 'react';
import * as imagesApi from '@/api/images.api';
import { useUIStore } from '@/store';
import { validateImageFile } from '@/utils/file/validation';

interface UploadProgress {
  total: number;
  completed: number;
  failed: number;
  current: string | null;
}

interface UseUploadImagesReturn {
  uploadImages: (projectId: string, files: File[]) => Promise<void>;
  isUploading: boolean;
  progress: UploadProgress;
  error: string | null;
  cancel: () => void;
}

export function useUploadImages(): UseUploadImagesReturn {
  const [isUploading, setIsUploading] = useState(false);
  const [progress, setProgress] = useState<UploadProgress>({
    total: 0, completed: 0, failed: 0, current: null
  });
  const [error, setError] = useState<string | null>(null);
  const abortRef = useRef<AbortController | null>(null);
  const showToast = useUIStore((s) => s.showToast);
  
  const uploadImages = useCallback(async (projectId: string, files: File[]) => {
    // 1. Validate all files first
    const validFiles: File[] = [];
    for (const file of files) {
      const result = validateImageFile(file);
      if (!result.valid) {
        showToast({ type: 'error', message: `${file.name}: ${result.error}` });
        continue;
      }
      validFiles.push(file);
    }
    
    if (validFiles.length === 0) {
      setError('No valid files to upload');
      return;
    }
    
    setIsUploading(true);
    setError(null);
    setProgress({ total: validFiles.length, completed: 0, failed: 0, current: null });
    abortRef.current = new AbortController();
    
    try {
      // 2. Get signed upload URLs from backend
      const uploadUrlsResponse = await imagesApi.getUploadUrls(projectId, {
        files: validFiles.map((f) => ({
          fileName: f.name,
          contentType: f.type,
          size: f.size,
        })),
      });
      
      // 3. Upload each file to Storage in parallel (with concurrency limit)
      const successfulIds: string[] = [];
      const CONCURRENT_UPLOADS = 5;
      
      for (let i = 0; i < uploadUrlsResponse.length; i += CONCURRENT_UPLOADS) {
        if (abortRef.current.signal.aborted) break;
        
        const batch = uploadUrlsResponse.slice(i, i + CONCURRENT_UPLOADS);
        const batchFiles = validFiles.slice(i, i + CONCURRENT_UPLOADS);
        
        await Promise.all(
          batch.map(async (urlInfo, idx) => {
            const file = batchFiles[idx];
            setProgress((p) => ({ ...p, current: file.name }));
            
            try {
              await imagesApi.uploadToSignedUrl(urlInfo.uploadUrl, file);
              successfulIds.push(urlInfo.imageId);
              setProgress((p) => ({ ...p, completed: p.completed + 1 }));
            } catch (err) {
              console.error(`Failed to upload ${file.name}:`, err);
              setProgress((p) => ({ ...p, failed: p.failed + 1 }));
            }
          })
        );
      }
      
      // 4. Confirm successful uploads with backend
      if (successfulIds.length > 0) {
        await imagesApi.confirmUpload(projectId, { imageIds: successfulIds });
        showToast({
          type: 'success',
          message: `Uploaded ${successfulIds.length} images`,
        });
      }
      
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Upload failed');
      showToast({ type: 'error', message: 'Upload failed' });
    } finally {
      setIsUploading(false);
      setProgress((p) => ({ ...p, current: null }));
      abortRef.current = null;
    }
  }, [showToast]);
  
  const cancel = useCallback(() => {
    abortRef.current?.abort();
    setIsUploading(false);
  }, []);
  
  return { uploadImages, isUploading, progress, error, cancel };
}
```

```typescript
  const [error, setError] = useState<string | null>(null);
  
  const masks = useMasksStore((s) => s.masks);
  const setDirty = useMasksStore((s) => s.setDirty);
  const setMasks = useMasksStore((s) => s.setMasks);
  const showToast = useUIStore((s) => s.showToast);
  
  const saveMasks = useCallback(async (
    projectId: string,
    imageId: string
  ): Promise<Mask[]> => {
    setIsLoading(true);
    setError(null);
    
    try {
      // Convert working masks to API format
      const input: SaveMasksInput = {
        masks: masks
          .filter((m) => m.classId !== null)  // Skip unclassified
          .map((m) => ({
            classId: m.classId!,
            data: {
              type: 'polygon' as const,
              polygon: m.polygon,
            },
            boundingBox: polygonToBBox(m.polygon[0]),
            area: polygonArea(m.polygon[0]),
            source: m.source,
          })),
      };
      
      // Save to backend
      const savedMasks = await labelingApi.saveMasks(projectId, imageId, input);
      
      // Update store with server-assigned IDs
      const workingMasks = savedMasks.map((m) => ({
        id: m.id,
        classId: m.classId,
        polygon: m.data.polygon,
        isSelected: false,
        isHovered: false,
        isDirty: false,
        source: m.source,

      setError(message);
      showToast({ type: 'error', message });
      throw err;
    } finally {
      setIsLoading(false);
    }
  }, [masks, setMasks, setDirty, showToast]);
```

### `frontend/src/hooks/canvas/useMaskEditor.ts`

**Purpose:** Mask manipulation operations for canvas.

**Implementation Notes:**
- Imports: `useMasksStore`, `useHistoryStore`, geometry utils
- Wraps store operations with undo/redo recording
- Provides high-level editing operations
- Used by: `LabelingCanvas.tsx`, `MaskList.tsx`

```typescript
import { useCallback } from 'react';
import { useMasksStore, useHistoryStore } from '@/store';
import { WorkingMask, Point } from '@/types';
import { generatePrefixedId } from '@/utils/helpers/generateId';
import { polygonArea, polygonToBBox } from '@/utils/geometry';

interface UseMaskEditorReturn {
  masks: WorkingMask[];
  selectedMask: WorkingMask | null;
  
  addMask: (polygon: Point[][], classId: string | null, source: string) => WorkingMask;
  updatePolygon: (maskId: string, polygon: Point[][]) => void;
  updateClass: (maskId: string, classId: string) => void;
  deleteMask: (maskId: string) => void;
  selectMask: (maskId: string | null) => void;
  duplicateSelected: () => WorkingMask | null;
  isDirty: () => boolean;
}

export function useMaskEditor(): UseMaskEditorReturn {
  const masks = useMasksStore((s) => s.masks);
  const selectedMaskId = useMasksStore((s) => s.selectedMaskId);
  const selectedMask = masks.find((m) => m.id === selectedMaskId) ?? null;
  
  const storeMethods = useMasksStore((s) => ({
    addMask: s.addMask,
    updateMask: s.updateMask,
    deleteMask: s.deleteMask,
    selectMask: s.selectMask,
  }));
  
  const pushAction = useHistoryStore((s) => s.pushAction);
  const isDirtyState = useMasksStore((s) => s.isDirty);
  
  /**
   * Add new mask with undo support
   */
  const addMask = useCallback((
    polygon: Point[][],
    classId: string | null,
    source: string
  ): WorkingMask => {
    const mask: WorkingMask = {
      id: generatePrefixedId('mask'),
      classId,
      polygon,
      isSelected: false,
      isHovered: false,
      isDirty: true,
      source,
    };
    
    // Record for undo
    pushAction({
      type: 'add_mask',
      maskId: mask.id,
      before: null,
      after: mask,
    });
    
    storeMethods.addMask(mask);
    return mask;
  }, [storeMethods, pushAction]);
  
  /**
   * Update mask polygon with undo support
   */
  const updatePolygon = useCallback((maskId: string, polygon: Point[][]) => {
    const before = masks.find((m) => m.id === maskId);
    if (!before) return;
    
    const after = { ...before, polygon };
    
    pushAction({
      type: 'modify_mask',
      maskId,
      before,
      after,
    });
    
    storeMethods.updateMask(maskId, { polygon });
  }, [masks, storeMethods, pushAction]);
  
  /**
   * Update mask class with undo support
   */
  const updateClass = useCallback((maskId: string, classId: string) => {
    const before = masks.find((m) => m.id === maskId);
    if (!before) return;
    
    pushAction({
      type: 'change_class',
      maskId,
      before,
      after: { ...before, classId },
    });
    
    storeMethods.updateMask(maskId, { classId });
  }, [masks, storeMethods, pushAction]);
  
  /**
   * Delete mask with undo support
   */
  const deleteMask = useCallback((maskId: string) => {
    const before = masks.find((m) => m.id === maskId);
    if (!before) return;
    
    pushAction({
      type: 'delete_mask',
      maskId,
      before,
      after: null,
    });
    
    storeMethods.deleteMask(maskId);
  }, [masks, storeMethods, pushAction]);
  
  /**
   * Duplicate currently selected mask
   */
  const duplicateSelected = useCallback((): WorkingMask | null => {
    if (!selectedMask) return null;
    
    // Offset the duplicate slightly
    const offsetPolygon = selectedMask.polygon.map((ring) =>
      ring.map((p) => ({ x: p.x + 20, y: p.y + 20 }))
    );
    
    return addMask(offsetPolygon, selectedMask.classId, 'manual');
  }, [selectedMask, addMask]);
  
  return {
    masks,
    selectedMask,
    addMask,
    updatePolygon,
    updateClass,
    deleteMask,
    selectMask: storeMethods.selectMask,
    duplicateSelected,
    isDirty: () => isDirtyState,
  };
}
```

### `backend/src/middleware/auth.middleware.ts`

**Purpose:** Verify Firebase ID token and extract user.

**Implementation Notes:**
- Imports: `auth` from firebase-admin, `UnauthorizedError` from utils
- Extracts token from `Authorization: Bearer <token>` header
- Verifies token with Firebase Admin SDK
- Attaches `req.user` with `{ uid, email }` for downstream use
- Used by: All protected routes (applied globally or per-route)

```typescript
import { Request, Response, NextFunction } from 'express';
import { auth } from '@/config/firebase';
import { UnauthorizedError } from '@/utils/errors';

// Extend Express Request type
declare global {
  namespace Express {
    interface Request {
      user?: {
        uid: string;
        email: string;
      };
    }
  }
}

/**
 * Verify Firebase ID token from Authorization header
 * Attaches user info to req.user
 * Throws UnauthorizedError if token invalid/missing
 */
export async function authMiddleware(
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> {
  try {
    const authHeader = req.headers.authorization;
    
    if (!authHeader?.startsWith('Bearer ')) {
      throw new UnauthorizedError('Missing authorization header');
    }
    
    const token = authHeader.slice(7);  // Remove "Bearer "
    
    // Verify with Firebase Admin
    const decodedToken = await auth.verifyIdToken(token);
    
    // Attach to request
    req.user = {
      uid: decodedToken.uid,
      email: decodedToken.email || '',
    };
    
    next();
  } catch (error) {
    next(new UnauthorizedError('Invalid or expired token'));
  }
}
```

### `backend/src/middleware/project.middleware.ts`

**Purpose:** Load project and verify membership.

**Implementation Notes:**
- Imports: `getProject` from project.model, `getMember` from member.model
- Requires `projectId` in route params
- Loads project and membership in parallel
- Attaches `req.project` and `req.membership` for downstream use
- Used by: All `/projects/:projectId/*` routes

```typescript
import { Request, Response, NextFunction } from 'express';
import { getProject } from '@/models/project.model';
import { getMember } from '@/models/member.model';
import { NotFoundError, ForbiddenError } from '@/utils/errors';
import { Project, ProjectMember } from '@shared/types';

// Extend Express Request type
declare global {
  namespace Express {
    interface Request {
      project?: Project;
      membership?: ProjectMember;
    }
  }
}

/**
 * Load project and verify user is a member
 * Requires projectId in params and user from authMiddleware
 */
export async function projectMiddleware(
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> {
  try {
    const { projectId } = req.params;
    const userId = req.user?.uid;
    
    if (!projectId) {
      throw new NotFoundError('Project', 'missing');
    }
    
    if (!userId) {
      throw new ForbiddenError('Authentication required');
    }
    
    // Load project and membership in parallel
    const [project, membership] = await Promise.all([
      getProject(projectId),
      getMember(projectId, userId),
    ]);
    
    if (!project) {
      throw new NotFoundError('Project', projectId);
    }
    
    if (!membership) {
      throw new ForbiddenError('Not a member of this project');
    }
    
    // Attach to request
    req.project = { ...project, id: projectId };
    req.membership = membership;
    
    next();
  } catch (error) {
    next(error);
  }
}
```

```typescript
/**
 * Format Zod errors into readable message
 */
function formatZodError(error: ZodError): string {
  return error.errors
    .map((e) => `${e.path.join('.')}: ${e.message}`)
    .join(', ');
}

/**
 * Validate request body against Zod schema
 * Replaces req.body with parsed data
 * 
 * @example
 * router.post('/projects', validate(createProjectSchema), createProject);
 */
export function validate(schema: ZodSchema) {
  return (req: Request, res: Response, next: NextFunction): void => {
    try {
      req.body = schema.parse(req.body);
      next();
    } catch (error) {
      if (error instanceof ZodError) {
        next(new ValidationError(formatZodError(error)));
      } else {
        next(error);
      }
    }
  };
}

/**
```
```typescript
 * Save masks - replaces ALL existing masks
 * Uses transaction for atomicity
 */
export async function saveMasks(
  projectId: string,
  imageId: string,
  input: SaveMasksInput,
  userId: string
): Promise<Mask[]> {
  const masksCollection = getMasksCollection(projectId, imageId);
  
  // Run in transaction
  const savedMasks = await db.runTransaction(async (transaction) => {
    // 1. Get all existing masks
    const existingSnapshot = await transaction.get(masksCollection);
    
    // 2. Delete all existing masks
    existingSnapshot.docs.forEach((doc) => {
      transaction.delete(doc.ref);
    });
    
    // 3. Create new masks
    const now = FieldValue.serverTimestamp();
    const newMasks: Mask[] = [];
    
    for (const maskInput of input.masks) {
      const docRef = masksCollection.doc();
      const mask: Omit<Mask, 'id'> = {
        ...maskInput,
        createdBy: userId,
        createdAt: now,
        updatedAt: now,
      };
      
      transaction.set(docRef, mask);
      newMasks.push({ id: docRef.id, ...mask } as Mask);
    }
    
    return newMasks;
  });
  
  // 4. Trigger combined mask generation (async, don't wait)
  generateCombinedMask(projectId, imageId).catch((err) => {
    console.error('Failed to generate combined mask:', err);
  });
  
  return savedMasks;
}
```

### `backend/functions/scheduledLockCleanup.ts`

**Purpose:** Expire stale locks hourly.

**Implementation Notes:**
- Imports: functions SDK, lock service
- Runs on schedule via Cloud Scheduler
- Queries all projects, finds expired locks
- Updates images to release locks
- Used by: Runs automatically on schedule

```typescript
import * as functions from 'firebase-functions';
import { db } from '@/config/firebase';

/**
 * Clean up expired locks across all projects
 * Runs every hour
 */
export const scheduledLockCleanup = functions.pubsub
  .schedule('every 1 hours')
  .onRun(async (context) => {
    const now = new Date();
    
    // Query all images with expired locks
    const expiredQuery = await db.collectionGroup('images')
      .where('lockState.locked', '==', true)
      .where('lockState.expiresAt', '<', now)
      .get();
    
    if (expiredQuery.empty) {
      functions.logger.info('No expired locks found');
      return;
    }
    
    // Batch update to release locks
    const batches: FirebaseFirestore.WriteBatch[] = [];
    let currentBatch = db.batch();
    let operationCount = 0;
    
    for (const doc of expiredQuery.docs) {
      currentBatch.update(doc.ref, {
        'lockState.locked': false,
      });
      
      operationCount++;
      
      // Firestore batch limit is 500
      if (operationCount >= 500) {
        batches.push(currentBatch);
        currentBatch = db.batch();
        operationCount = 0;
      }
    }
    
    if (operationCount > 0) {
      batches.push(currentBatch);
    }
    
    // Execute all batches
    await Promise.all(batches.map((batch) => batch.commit()));
    
    functions.logger.info(`Released ${expiredQuery.size} expired locks`);
  });
```

### `sam3-service/src/inference/postprocess.py`

**Purpose:** Mask post-processing utilities.

**Implementation Notes:**
- Imports: cv2 for contour detection, numpy
- Converts binary mask to polygon via contour tracing
- Simplifies polygon to reduce point count
- Used by: all inference modules

```python
from typing import List, Dict, Any
import numpy as np
import cv2

def mask_to_polygon(
    mask: np.ndarray,
    simplify_tolerance: float = 2.0
) -> List[List[Dict[str, float]]]:
    """
    Convert binary mask to polygon coordinates
    
    Args:
        mask: Binary mask array (H, W) with values 0 or 1
        simplify_tolerance: Douglas-Peucker simplification tolerance
    
    Returns:
        List of rings, each ring is list of {x, y} points
        First ring is outer boundary, rest are holes
    """
    # Ensure binary mask
    mask_uint8 = (mask > 0).astype(np.uint8) * 255
    
    # Find contours
    contours, hierarchy = cv2.findContours(
        mask_uint8,
        cv2.RETR_CCOMP,  # Two-level hierarchy (outer + holes)
        cv2.CHAIN_APPROX_SIMPLE
    )
    
    if not contours:
        return []
    
    rings = []
    
    for i, contour in enumerate(contours):
        # Simplify contour
        epsilon = simplify_tolerance
        simplified = cv2.approxPolyDP(contour, epsilon, closed=True)
        
        # Skip if too few points
        if len(simplified) < 3:
            continue
        
        # Convert to list of {x, y} dicts
        ring = [
            {"x": float(pt[0][0]), "y": float(pt[0][1])}
            for pt in simplified
        ]
        
        rings.append(ring)
    
    return rings

def calculate_bbox(mask: np.ndarray) -> Dict[str, float]:
    """Calculate bounding box from mask"""
    rows = np.any(mask, axis=1)
    cols = np.any(mask, axis=0)
    
    if not rows.any():
        return {"x": 0, "y": 0, "w": 0, "h": 0}
    
    y_min, y_max = np.where(rows)[0][[0, -1]]
    x_min, x_max = np.where(cols)[0][[0, -1]]
    
    return {
        "x": float(x_min),
        "y": float(y_min),
        "w": float(x_max - x_min + 1),
        "h": float(y_max - y_min + 1)
    }

def calculate_area(mask: np.ndarray) -> int:
    """Calculate mask area in pixels"""
    return int(np.sum(mask > 0))
```

### Polygon Editor Hook

```typescript
// frontend/src/hooks/canvas/usePolygonEditor.ts

/**
 * Purpose: High-level polygon editing operations with undo support
 * 
 * Implementation Notes:
 * - Imports: useMasksStore, useToolStore, useHistoryStore, geometry utils
 * - All operations wrapped with history recording
 * - Validates operations (min 3 vertices, etc.)
 * - Used by: LabelingCanvas.tsx for polygon edit mode
 */

import { useCallback } from 'react';
import { useMasksStore, useToolStore, useHistoryStore } from '@/store';
import { Point } from '@/types';
import { pointToLineDistance, insertPointInRing } from '@/utils/geometry';

interface UsePolygonEditorReturn {
  // State
  isEditing: boolean;
  selectedMaskId: string | null;
  editState: PolygonEditState;
  
  // Vertex operations
  moveVertex: (maskId: string, ringIndex: number, vertexIndex: number, newPosition: Point) => void;
  addVertex: (maskId: string, ringIndex: number, edgeIndex: number, position: Point) => void;
  deleteVertex: (maskId: string, ringIndex: number, vertexIndex: number) => boolean;
  
  // Drag operations (for smooth dragging)
  startDrag: (maskId: string, ringIndex: number, vertexIndex: number) => void;
  updateDrag: (position: Point) => void;
  endDrag: () => void;
  
  // Helpers
  findNearestVertex: (position: Point, threshold: number) => { maskId: string; ringIndex: number; vertexIndex: number } | null;
  findNearestEdge: (position: Point, threshold: number) => { maskId: string; ringIndex: number; edgeIndex: number; point: Point } | null;
}

export function usePolygonEditor(): UsePolygonEditorReturn {
  const masks = useMasksStore((s) => s.masks);
  const selectedMaskId = useMasksStore((s) => s.selectedMaskId);
  const updateMask = useMasksStore((s) => s.updateMask);
  
  const editState = useToolStore((s) => s.polygonEdit);
  const { startVertexDrag, endVertexDrag, resetPolygonEdit } = useToolStore();
  
  const pushAction = useHistoryStore((s) => s.pushAction);
  
  /**
   * Move a vertex to a new position
   */
  const moveVertex = useCallback((
    maskId: string,
    ringIndex: number,
    vertexIndex: number,
    newPosition: Point
  ) => {
    const mask = masks.find((m) => m.id === maskId);
    if (!mask) return;
    
    const oldPolygon = mask.polygon;
    const newPolygon = oldPolygon.map((ring, rIdx) => {
      if (rIdx !== ringIndex) return ring;
      return ring.map((point, pIdx) => 
        pIdx === vertexIndex ? newPosition : point
      );
    });
    
    // Record for undo
    pushAction({
      type: 'modify_mask',
      maskId,
      before: { ...mask },
      after: { ...mask, polygon: newPolygon },
    });
    
    updateMask(maskId, { polygon: newPolygon });
  }, [masks, updateMask, pushAction]);
  
  /**
   * Add a vertex at a position on an edge
   */
  const addVertex = useCallback((
    maskId: string,
    ringIndex: number,
    edgeIndex: number,
    position: Point
  ) => {
    const mask = masks.find((m) => m.id === maskId);
    if (!mask) return;
    
    const oldPolygon = mask.polygon;
    const newPolygon = oldPolygon.map((ring, rIdx) => {
      if (rIdx !== ringIndex) return ring;
      // Insert after edgeIndex
      const newRing = [...ring];
      newRing.splice(edgeIndex + 1, 0, position);
      return newRing;
    });
    
    pushAction({
      type: 'modify_mask',
      maskId,
      before: { ...mask },
      after: { ...mask, polygon: newPolygon },
    });
    
    updateMask(maskId, { polygon: newPolygon });
  }, [masks, updateMask, pushAction]);
  
  /**
   * Delete a vertex (if more than 3 remain)
   * Returns false if deletion not allowed
   */
  const deleteVertex = useCallback((
    maskId: string,
    ringIndex: number,
    vertexIndex: number
  ): boolean => {
    const mask = masks.find((m) => m.id === maskId);
    if (!mask) return false;
    
    const ring = mask.polygon[ringIndex];
    if (ring.length <= 3) {
      // Can't have less than 3 vertices in a polygon
      return false;
    }
    
    const oldPolygon = mask.polygon;
    const newPolygon = oldPolygon.map((r, rIdx) => {
      if (rIdx !== ringIndex) return r;
      return r.filter((_, pIdx) => pIdx !== vertexIndex);
    });
    
    pushAction({
      type: 'modify_mask',
      maskId,
      before: { ...mask },
      after: { ...mask, polygon: newPolygon },
    });
    
    updateMask(maskId, { polygon: newPolygon });
    return true;
  }, [masks, updateMask, pushAction]);
  
  /**
   * Start dragging a vertex (for smooth drag preview)
   */
  const startDrag = useCallback((
    maskId: string,
    ringIndex: number,
    vertexIndex: number
  ) => {
    const mask = masks.find((m) => m.id === maskId);
    if (!mask) return;
    
    startVertexDrag(vertexIndex, ringIndex, mask.polygon);
  }, [masks, startVertexDrag]);
  
  /**
   * Update position during drag (preview only, no history)
   */
  const updateDrag = useCallback((position: Point) => {
    const { draggingVertexIndex, activeRingIndex, dragStartPolygon } = editState;
    if (draggingVertexIndex === null || !selectedMaskId || !dragStartPolygon) return;
    
    // Update polygon without recording history (will record on endDrag)
    const newPolygon = dragStartPolygon.map((ring, rIdx) => {
      if (rIdx !== activeRingIndex) return ring;
      return ring.map((point, pIdx) =>
        pIdx === draggingVertexIndex ? position : point
      );
    });
    
    // Direct update without history
    useMasksStore.getState().updateMask(selectedMaskId, { polygon: newPolygon });
  }, [editState, selectedMaskId]);
  
  /**
   * End drag and record to history
   */
  const endDrag = useCallback(() => {
    const { draggingVertexIndex, dragStartPolygon } = editState;
    if (draggingVertexIndex === null || !selectedMaskId || !dragStartPolygon) return;
    
    const mask = masks.find((m) => m.id === selectedMaskId);
    if (!mask) return;
    
    // Record the complete change
    pushAction({
      type: 'modify_mask',
      maskId: selectedMaskId,
      before: { ...mask, polygon: dragStartPolygon },
      after: { ...mask },
    });
    
    endVertexDrag();
  }, [editState, selectedMaskId, masks, pushAction, endVertexDrag]);
  
  /**
   * Find nearest vertex to a point within threshold
   */
  const findNearestVertex = useCallback((
    position: Point,
    threshold: number
  ): { maskId: string; ringIndex: number; vertexIndex: number } | null => {
    let nearest: { maskId: string; ringIndex: number; vertexIndex: number; distance: number } | null = null;
    
    for (const mask of masks) {
      for (let ringIndex = 0; ringIndex < mask.polygon.length; ringIndex++) {
        const ring = mask.polygon[ringIndex];
        for (let vertexIndex = 0; vertexIndex < ring.length; vertexIndex++) {
          const vertex = ring[vertexIndex];
          const distance = Math.hypot(vertex.x - position.x, vertex.y - position.y);
          
          if (distance <= threshold && (!nearest || distance < nearest.distance)) {
            nearest = { maskId: mask.id, ringIndex, vertexIndex, distance };
          }
        }
      }
    }
    
    return nearest ? { maskId: nearest.maskId, ringIndex: nearest.ringIndex, vertexIndex: nearest.vertexIndex } : null;
  }, [masks]);
  
  /**
   * Find nearest edge to a point within threshold
   * Returns the closest point on the edge as well
   */
  const findNearestEdge = useCallback((
    position: Point,
    threshold: number
  ): { maskId: string; ringIndex: number; edgeIndex: number; point: Point } | null => {
    let nearest: { maskId: string; ringIndex: number; edgeIndex: number; point: Point; distance: number } | null = null;
    
    for (const mask of masks) {
      for (let ringIndex = 0; ringIndex < mask.polygon.length; ringIndex++) {
        const ring = mask.polygon[ringIndex];
        for (let i = 0; i < ring.length; i++) {
          const p1 = ring[i];
          const p2 = ring[(i + 1) % ring.length];
          
          const { distance, point } = pointToLineDistance(position, p1, p2);
          
          if (distance <= threshold && (!nearest || distance < nearest.distance)) {
            nearest = { maskId: mask.id, ringIndex, edgeIndex: i, point, distance };
          }
        }
      }
    }
    
    return nearest ? { maskId: nearest.maskId, ringIndex: nearest.ringIndex, edgeIndex: nearest.edgeIndex, point: nearest.point } : null;
  }, [masks]);
  
  return {
    isEditing: selectedMaskId !== null,
    selectedMaskId,
    editState,
    moveVertex,
    addVertex,
    deleteVertex,
    startDrag,
    updateDrag,
    endDrag,
    findNearestVertex,
    findNearestEdge,
  };
}
```

```typescript
// frontend/src/utils/geometry/polygon.ts (additions for editing)

/**
 * Purpose: Polygon geometry utilities for editing
 * 
 * Implementation Notes:
 * - Pure functions, no side effects
 * - All coordinates in image space
 * - Used by: usePolygonEditor, LabelingCanvas
 */

/**
 * Calculate distance from point to line segment
 * Returns distance and closest point on segment
 */
export function pointToLineDistance(
  point: Point,
  lineStart: Point,
  lineEnd: Point
): { distance: number; point: Point } {
  const dx = lineEnd.x - lineStart.x;
  const dy = lineEnd.y - lineStart.y;
  const lengthSq = dx * dx + dy * dy;
  
  if (lengthSq === 0) {
    // Line segment is a point
    return {
      distance: Math.hypot(point.x - lineStart.x, point.y - lineStart.y),
      point: lineStart,
    };
  }
  
  // Project point onto line, clamped to segment
  let t = ((point.x - lineStart.x) * dx + (point.y - lineStart.y) * dy) / lengthSq;
  t = Math.max(0, Math.min(1, t));
  
  const closestPoint = {
    x: lineStart.x + t * dx,
    y: lineStart.y + t * dy,
  };
  
  return {
    distance: Math.hypot(point.x - closestPoint.x, point.y - closestPoint.y),
    point: closestPoint,
  };
}

/**
 * Check if point is inside polygon (ray casting)
 */
export function pointInPolygon(point: Point, polygon: Point[]): boolean {
  let inside = false;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    const xi = polygon[i].x, yi = polygon[i].y;
    const xj = polygon[j].x, yj = polygon[j].y;
    
    if (((yi > point.y) !== (yj > point.y)) &&
        (point.x < (xj - xi) * (point.y - yi) / (yj - yi) + xi)) {
      inside = !inside;
    }
  }
  return inside;
}

/**
 * Simplify polygon using Douglas-Peucker algorithm
 */
export function simplifyPolygon(points: Point[], tolerance: number): Point[] {
  if (points.length <= 2) return points;
  
  // Find point with max distance from line between first and last
  let maxDistance = 0;
  let maxIndex = 0;
  
  const first = points[0];
  const last = points[points.length - 1];
  
  for (let i = 1; i < points.length - 1; i++) {
    const { distance } = pointToLineDistance(points[i], first, last);
    if (distance > maxDistance) {
      maxDistance = distance;
      maxIndex = i;
    }
  }
  
  if (maxDistance > tolerance) {
    // Recursively simplify
    const left = simplifyPolygon(points.slice(0, maxIndex + 1), tolerance);
    const right = simplifyPolygon(points.slice(maxIndex), tolerance);
    return [...left.slice(0, -1), ...right];
  }
  
  return [first, last];
}

/**
 * Calculate polygon centroid
 */
export function polygonCentroid(points: Point[]): Point {
  let cx = 0, cy = 0;
  for (const p of points) {
    cx += p.x;
    cy += p.y;
  }
  return { x: cx / points.length, y: cy / points.length };
}
```

```typescript
/**
 * Render polygon with editable vertices
 * Called when tool is 'polygon_edit' and mask is selected
 */
function renderEditablePolygon(
  ctx: CanvasRenderingContext2D,
  polygon: Point[][],
  viewport: Viewport,
  editState: PolygonEditState,
  classColor: string
): void {
  const { hoveredVertexIndex, hoveredEdgeIndex, draggingVertexIndex } = editState;
  
  // Transform points to screen coordinates
  const toScreen = (p: Point) => ({
    x: p.x * viewport.zoom + viewport.x,
    y: p.y * viewport.zoom + viewport.y,
  });
  
  for (let ringIndex = 0; ringIndex < polygon.length; ringIndex++) {
    const ring = polygon[ringIndex];
    const screenRing = ring.map(toScreen);
    
    // Draw polygon fill (semi-transparent)
    ctx.fillStyle = classColor + '40';  // 25% opacity
    ctx.beginPath();
    ctx.moveTo(screenRing[0].x, screenRing[0].y);
    for (let i = 1; i < screenRing.length; i++) {
      ctx.lineTo(screenRing[i].x, screenRing[i].y);
    }
    ctx.closePath();
    ctx.fill();
    
    // Draw edges
    ctx.strokeStyle = classColor;
    ctx.lineWidth = 2;
    ctx.beginPath();
    for (let i = 0; i < screenRing.length; i++) {
      const p1 = screenRing[i];
      const p2 = screenRing[(i + 1) % screenRing.length];
      
      ctx.moveTo(p1.x, p1.y);
      ctx.lineTo(p2.x, p2.y);
      
      // Highlight hovered edge
      if (hoveredEdgeIndex === i && ringIndex === editState.activeRingIndex) {
        ctx.save();
        ctx.strokeStyle = '#FFD700';  // Gold highlight
        ctx.lineWidth = 4;
        ctx.stroke();
        ctx.restore();
        
        // Draw add-vertex indicator at midpoint
        const mid = { x: (p1.x + p2.x) / 2, y: (p1.y + p2.y) / 2 };
        ctx.fillStyle = '#FFD700';
        ctx.beginPath();
        ctx.arc(mid.x, mid.y, 4, 0, Math.PI * 2);
        ctx.fill();
      }
    }
    ctx.stroke();
    
    // Draw vertices
    for (let i = 0; i < screenRing.length; i++) {
      const p = screenRing[i];
      const isHovered = hoveredVertexIndex === i && ringIndex === editState.activeRingIndex;
      const isDragging = draggingVertexIndex === i && ringIndex === editState.activeRingIndex;
      
      // Vertex circle
      ctx.fillStyle = isDragging ? '#FF6B6B' : (isHovered ? '#4ECDC4' : '#FFFFFF');
      ctx.strokeStyle = classColor;
      ctx.lineWidth = 2;
      
      const radius = isHovered || isDragging ? 8 : 6;
      
      ctx.beginPath();
      ctx.arc(p.x, p.y, radius, 0, Math.PI * 2);
      ctx.fill();
      ctx.stroke();
    }
  }
}
```
